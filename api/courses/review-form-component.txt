<script>
(function() {
    const BASE = '/proj';
    let selectedRating = 0;
    const courseId = getCourseIdFromURL(); // You need to implement this based on your URL structure

    // Star rating interaction
    const stars = document.querySelectorAll('.star');
    const ratingValue = document.getElementById('ratingValue');

    stars.forEach(star => {
        star.addEventListener('click', function() {
            selectedRating = parseInt(this.getAttribute('data-rating'));
            ratingValue.textContent = selectedRating;
            updateStars(selectedRating);
        });

        star.addEventListener('mouseenter', function() {
            const hoverRating = parseInt(this.getAttribute('data-rating'));
            updateStars(hoverRating);
        });
    });

    document.getElementById('starRating').addEventListener('mouseleave', function() {
        updateStars(selectedRating);
    });

    function updateStars(rating) {
        stars.forEach((star, index) => {
            if (index < rating) {
                star.classList.add('active');
            } else {
                star.classList.remove('active');
            }
        });
    }

    // Submit review
    document.getElementById('submitReviewBtn').addEventListener('click', async function() {
        const comment = document.getElementById('reviewComment').value.trim();
        const messageEl = document.getElementById('reviewMessage');
        const submitBtn = this;

        if (selectedRating === 0) {
            showMessage('Please select a rating', 'error');
            return;
        }

        if (!comment) {
            showMessage('Please write a comment', 'error');
            return;
        }

        try {
            submitBtn.disabled = true;
            showMessage('Submitting review...', 'success');

            const response = await fetch(`${BASE}/api/reviews/submit.php`, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                credentials: 'include',
                body: JSON.stringify({
                    courseId: courseId,
                    rating: selectedRating,
                    comment: comment
                })
            });

            const data = await response.json();

            if (!response.ok || !data.ok) {
                throw new Error(data.error || 'Failed to submit review');
            }

            showMessage(data.message || 'Review submitted successfully!', 'success');
            document.getElementById('reviewComment').value = '';
            selectedRating = 0;
            updateStars(0);
            ratingValue.textContent = '0';

            // Reload reviews
            setTimeout(() => loadReviews(), 1000);

        } catch (error) {
            showMessage(error.message, 'error');
        } finally {
            submitBtn.disabled = false;
        }
    });

    function showMessage(text, type) {
        const messageEl = document.getElementById('reviewMessage');
        messageEl.textContent = text;
        messageEl.className = `review-message ${type}`;
        messageEl.style.display = 'block';

        if (type === 'success') {
            setTimeout(() => {
                messageEl.style.display = 'none';
            }, 5000);
        }
    }

    // Load existing reviews
    async function loadReviews() {
        const reviewsList = document.getElementById('reviewsList');

        try {
            const response = await fetch(`${BASE}/api/reviews/by-course.php?courseId=${courseId}`, {
                credentials: 'include'
            });
            const data = await response.json();

            if (!data.ok || !data.reviews || data.reviews.length === 0) {
                reviewsList.innerHTML = '<p style="color: #777; text-align: center;">No reviews yet. Be the first to review!</p>';
                return;
            }

            reviewsList.innerHTML = data.reviews.map(review => `
                <div class="review-card">
                    <div class="review-header">
                        <span class="reviewer-name">${escapeHtml(review.name || 'Anonymous')}</span>
                        <span class="review-rating">${'★'.repeat(review.rating)}${'☆'.repeat(5 - review.rating)}</span>
                    </div>
                    <p class="review-comment">${escapeHtml(review.comment)}</p>
                </div>
            `).join('');

        } catch (error) {
            reviewsList.innerHTML = '<p style="color: #a03030;">Failed to load reviews</p>';
        }
    }

    function getCourseIdFromURL() {
        const params = new URLSearchParams(window.location.search);
        return parseInt(params.get('id') || params.get('courseId') || '0');
    }

    function escapeHtml(text) {
        const div = document.createElement('div');
        div.textContent = text;
        return div.innerHTML;
    }

    // Load reviews on page load
    if (courseId) {
        loadReviews();
    } else {
        document.getElementById('reviewSection').innerHTML = '<p style="color: #a03030;">Invalid course</p>';
    }
})();
</script>
